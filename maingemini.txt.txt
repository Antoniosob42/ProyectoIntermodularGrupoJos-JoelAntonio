#  RELOJ
# ---------------------------------------------------------------------
_OFF_EPOCH = 946684800 if time.gmtime(0)[0] == 2000 else 0
_ref_ms = None
_ref_ticks = 0


def hora_valida():
    try:
        return (time.time() + _OFF_EPOCH) > 1700000000
    except Exception:
        return False


def sync_ref():
    global _ref_ms, _ref_ticks
    _ref_ticks = time.ticks_ms()
    _ref_ms = int((time.time() + _OFF_EPOCH) * 1000)


def timestamp_ms():
    try:
        if _ref_ms is None:
            if hora_valida():
                sync_ref()
            else:
                return int(time.ticks_ms())
        return _ref_ms + time.ticks_diff(time.ticks_ms(), _ref_ticks)
    except Exception:
        return int(time.ticks_ms())


def hora_texto():
    try:
        t = time.gmtime(time.time() + int(CFG["utc_offset_h"] * 3600))
        return "%02d:%02d:%02d" % (t[3], t[4], t[5])
    except Exception:
        return "--:--:--"


def fmt_t(ms):
    if ms is None:
        return "--:--.--"
    return "%d:%05.2f" % (ms // 60000, (ms % 60000) / 1000)

# ---------------------------------------------------------------------
#  GEOMETRÍA
# ---------------------------------------------------------------------
def a_metros(lat, lng, lat0, lng0):
    x = (lng - lng0) * 111320.0 * math.cos(math.radians(lat0))
    y = (lat - lat0) * 111320.0
    return x, y


def dist_m(lat1, lng1, lat2, lng2):
    x, y = a_metros(lat2, lng2, lat1, lng1)
    return math.sqrt(x * x + y * y)


def dist_seg(prev, cur, meta):
    ax, ay = a_metros(prev[0], prev[1], meta[0], meta[1])
    bx, by = a_metros(cur[0], cur[1], meta[0], meta[1])
    dx = bx - ax
    dy = by - ay
    l2 = dx * dx + dy * dy
    if l2 < 1e-6:
        return math.sqrt(ax * ax + ay * ay)
    t = -(ax * dx + ay * dy) / l2
    if t < 0:
        t = 0
    elif t > 1:
        t = 1
    px = ax + t * dx
    py = ay + t * dy
    return math.sqrt(px * px + py * py)

# ---------------------------------------------------------------------
def firebase_request(metodo, ruta, datos=None):
    url = CFG["base_url"] + "/" + ruta + ".json"
    respuesta = None
    try:
        cuerpo = json.dumps(datos) if datos is not None else None
        if metodo == "POST":
            respuesta = requests.post(url, data=cuerpo)
        elif metodo == "PUT":
            respuesta = requests.put(url, data=cuerpo)
        elif metodo == "PATCH":
            respuesta = requests.patch(url, data=cuerpo)
        elif metodo == "GET":
            respuesta = requests.get(url)
        else:
            return False, None
        correcto = 200 <= respuesta.status_code < 300
        resultado = respuesta.json() if correcto and metodo in ("POST", "GET") else None
        return correcto, resultado
    except Exception as e:
        print("Error de Firebase:", e)
        return False, None
    finally:
        if respuesta is not None:
            try:
                respuesta.close()
            except Exception:
                pass
        gc.collect()

class Cola:
    def __init__(self, base, usa_sd):
        self.lock = _thread.allocate_lock()
        self.ram = []
        self.sd = usa_sd
        self.enviando = False
        self.perdidos = 0
        self.pos = 0
        self.n_sd = 0
        self.f_data = (base or "") + "/pendientes.jsonl"
        self.f_pos = (base or "") + "/pendientes.pos"
        if self.sd:
            self._cargar()

    def _cargar(self):
        try:
            try:
                with open(self.f_pos) as f:
                    self.pos = int(f.read().strip() or 0)
            except Exception:
                self.pos = 0
            n = 0
            with open(self.f_data) as f:
                f.seek(self.pos)
                while f.readline():
                    n += 1
            self.n_sd = n
        except Exception:
            self.pos = 0
            self.n_sd = 0
        if self.n_sd:
            print("Pendientes en SD:", self.n_sd)

    def total(self):
        return len(self.ram) + self.n_sd

    def add(self, ruta, valor):
        with self.lock:
            self.ram.append({"p": ruta, "d": valor})
            if self.enviando:
                return
            if self.sd:
                if len(self.ram) > CFG["ram_max"]:
                    self._spill()
            else:
                while len(self.ram) > CFG["ram_max_sin_sd"]:
                    self.ram.pop(0)
                    self.perdidos += 1

    def _spill(self):
        try:
            with open(self.f_data, "a") as f:
                for r in self.ram:
                    f.write(json.dumps(r))
                    f.write("\n")
            self.n_sd += len(self.ram)
            self.ram = []
        except Exception as e:
            print("Error escribiendo SD:", e)
            while len(self.ram) > CFG["ram_max_sin_sd"]:
                self.ram.pop(0)
                self.perdidos += 1

    def volcar(self):
        with self.lock:
            if self.sd and self.ram:
                self._spill()

    def _limpiar_sd(self):
        for ruta in (self.f_data, self.f_pos):
            try:
                os.remove(ruta)
            except Exception:
                pass
        self.pos = 0
        self.n_sd = 0

    def tomar(self, n):
        with self.lock:
            if self.sd and self.n_sd > 0:
                lote = []
                nueva = self.pos
                leidas = 0
                try:
                    with open(self.f_data) as f:
                        f.seek(self.pos)
                        while leidas < n:
                            ln = f.readline()
                            if not ln or not ln.endswith("\n"):
                                break
                            nueva = f.tell()
                            leidas += 1
                            try:
                                lote.append(json.loads(ln))
                            except Exception:
                                pass
                except Exception as e:
                    print("Error leyendo SD:", e)
                if leidas > 0:
                    return "sd", lote, (nueva, leidas)
                self._limpiar_sd()
            if self.ram:
                self.enviando = True
                lote = self.ram[:n]
                return "ram", lote, len(lote)
            return None, [], 0

    def confirmar(self, origen, info):
        with self.lock:
            if origen == "sd":
                nueva, leidas = info
                self.pos = nueva
                self.n_sd -= leidas
                if self.n_sd <= 0:
                    self._limpiar_sd()
                else:
                    try:
                        with open(self.f_pos, "w") as f:
                            f.write(str(self.pos))
                    except Exception as e:
                        print("Error guardando posición:", e)
            elif origen == "ram":
                del self.ram[:info]
                self.enviando = False

    def liberar(self):
        with self.lock:
            self.enviando = False

#  GPS REAL (NMEA por UART)
# ---------------------------------------------------------------------
class GpsReal:
    def __init__(self):
        from machine import UART
        self.uart = UART(CFG["gps_uart"], baudrate=CFG["gps_baud"],
                         tx=CFG["gps_tx"], rx=CFG["gps_rx"])
        self.buf = b""
        self.lat = 0.0
        self.lng = 0.0
        self.vel = 0.0
        self.rumbo = 0.0
        self.alt = 0.0
        self.sats = 0
        self.fix = False
        self.t_fix = 0

    def leer(self):
        try:
            n = self.uart.any()
            if not n:
                return
            d = self.uart.read(n)
            if d:
                self.buf += d
            if len(self.buf) > 1024:
                self.buf = self.buf[-512:]
            while True:
                i = self.buf.find(b"\n")
                if i < 0:
                    break
                linea = self.buf[:i]
                self.buf = self.buf[i + 1:]
                self._parsear(linea)
        except Exception as e:
            print("Error GPS:", e)

    @staticmethod
    def _grados(v, hemi):
        pt = v.find(".")
        g = float(v[:pt - 2]) + float(v[pt - 2:]) / 60.0
        return -g if hemi in ("S", "W") else g

    def _parsear(self, linea):
        try:
            s = linea.decode().strip()
        except Exception:
            return
        if not s.startswith("$") or "*" not in s:
            return
        cuerpo, _, cs = s[1:].partition("*")
        x = 0
        for ch in cuerpo:
            x ^= ord(ch)
        if ("%02X" % x) != cs[:2].upper():
            return
        c = cuerpo.split(",")
        tipo = c[0][2:]
        try:
            if tipo == "RMC" and len(c) > 8:
                if c[2] == "A" and c[3] and c[5]:
                    self.lat = self._grados(c[3], c[4])
                    self.lng = self._grados(c[5], c[6])
                    self.vel = (float(c[7]) if c[7] else 0.0) * 1.852
                    if c[8]:
                        self.rumbo = float(c[8])
                    self.fix = True
                    self.t_fix = time.ticks_ms()
                else:
                    self.fix = False
            elif tipo == "GGA" and len(c) > 9:
                self.sats = int(c[7]) if c[7] else 0
                self.alt = float(c[9]) if c[9] else 0.0
        except Exception:
            pass

    def valido(self):
        return self.fix and time.ticks_diff(time.ticks_ms(), self.t_fix) < 3000

class Circuito:
    def __init__(self):
        random.seed(timestamp_ms() & 0x3FFFFFFF)
        self.R0 = float(CFG["radio_circuito_m"])
        self.arm = []
        for k in (2, 3, 5):
            self.arm.append((k, random.uniform(0.05, 0.15), random.uniform(0, 2 * math.pi)))
        self.theta = 0.0

    def radio(self, th):
        r = 1.0
        for k, a, p in self.arm:
            r += a * math.cos(k * th + p)
        return self.R0 * r

    def avanzar(self, ds):
        r = self.radio(self.theta)
        self.theta = (self.theta + ds / max(r, 50.0)) % (2 * math.pi)

    def latlng(self):
        r = self.radio(self.theta)
        x = r * math.cos(self.theta)
        y = r * math.sin(self.theta)
        lat0 = CFG["centro_lat"]
        return (lat0 + y / 111320.0,
                CFG["centro_lng"] + x / (111320.0 * math.cos(math.radians(lat0))))


# ---------------------------------------------------------------------
#  DINÁMICA: velocidad por IMU
# ---------------------------------------------------------------------
def leer_imu():
    try:
        a = M5.Imu.getAccel()
        return (a[0], a[1], a[2])
    except Exception:
        return (0.0, 0.0, 0.0)


class Dinamica:
    def __init__(self):
        self.bias = [0.0, 0.0, 1.0]
        self.a = [0.0, 0.0, 1.0]
        self.v = 0.0
        self.vib = 0.0
        self.quiet = 0
        self.dev = 0.0
        self.rpm = float(CFG["rpm_ralenti"])

    def calibrar(self, muestras=60):
        s = [0.0, 0.0, 0.0]
        for _ in range(muestras):
            M5.update()
            r = leer_imu()
            for i in range(3):
                s[i] += r[i]
            time.sleep_ms(15)
        for i in range(3):
            self.bias[i] = s[i] / muestras
            self.a[i] = self.bias[i]

    def actualizar(self, dt, gps_ms=None):
        raw = leer_imu()
        e = CFG["eje_long"]
        for i in range(3):
            self.a[i] += 0.2 * (raw[i] - self.a[i])
        self.vib += 0.05 * (abs(raw[e] - self.a[e]) - self.vib)
        dev = (self.a[e] - self.bias[e]) * CFG["signo_long"]
        self.dev = dev

        parado = abs(dev) < 0.03 and self.vib < CFG["umbral_vib_g"]
        if gps_ms is not None and gps_ms < 0.3:
            parado = True
        self.quiet = self.quiet + 1 if parado else 0

        if self.quiet >= 25:
            self.v = 0.0
            for i in range(3):
                self.bias[i] += 0.01 * (self.a[i] - self.bias[i])
        else:
            self.v += dev * 9.81 * dt
            self.v -= self.v * 0.02 * dt
            if self.v < 0:
                self.v = 0.0
            elif self.v > 100:
                self.v = 100.0

        if gps_ms is not None:
            self.v += 0.15 * (gps_ms - self.v)

        objetivo = (CFG["rpm_ralenti"] + abs(dev) * CFG["rpm_k_acc"]
                    + self.v * 3.6 * CFG["rpm_k_vel"])
        self.rpm += 0.1 * (objetivo - self.rpm)
        if self.rpm > CFG["rpm_max"]:
            self.rpm = float(CFG["rpm_max"])
        return self.v

#  VUELTAS, SECTORES Y DELTA
# ---------------------------------------------------------------------
class Vueltas:
    def __init__(self):
        self.meta = None
        self.reset()

    def reset(self):
        self.num = 0
        self.t0 = 0
        self.ult_cruce = 0
        self.dist = 0.0
        self.armado = False
        self.limites = None
        self.sec_i = 0
        self.cum = []
        self.traza = []
        self.t_traza = 0
        self.mejor = None
        self.mejor_cum = None
        self.mejores_sec = None
        self.ultima = None
        self.ult_secs = None
        self.delta = None

    def fijar_meta(self, lat, lng, ahora):
        self.meta = (lat, lng)
        self.reset()
        self.t0 = ahora
        self.ult_cruce = ahora
        self.t_traza = ahora

    def teorica(self):
        return sum(self.mejores_sec) if self.mejores_sec else None

    def _delta(self, i, t):
        if self.mejor_cum and i < len(self.mejor_cum) - 1:
            self.delta = t - self.mejor_cum[i]

    def _definir_sectores(self, total_ms):
        n = CFG["n_sectores"]
        if n < 2 or len(self.traza) < 3 or self.dist < 1:
            self.limites = []
            self.cum = []
            return
        pts = [(0, 0)] + self.traza + [(int(self.dist), total_ms)]
        self.limites = [self.dist * (i + 1) / n for i in range(n - 1)]
        cum = []
        for lim in self.limites:
            t = total_ms
            for j in range(1, len(pts)):
                d0, t0 = pts[j - 1]
                d1, t1 = pts[j]
                if d0 <= lim <= d1:
                    t = t0 + (lim - d0) * (t1 - t0) / max(d1 - d0, 1)
                    break
            cum.append(int(t))
        self.cum = cum
        self.traza = []

    def _cerrar(self, ahora):
        t = time.ticks_diff(ahora, self.t0)
        self.num += 1
        if self.limites is None:
            self._definir_sectores(t)
        if self.limites is not None and len(self.cum) == len(self.limites):
            cum = self.cum + [t]
        else:
            cum = [t]
        secs = [cum[0]] + [cum[i] - cum[i - 1] for i in range(1, len(cum))]
        self.delta = (t - self.mejor) if self.mejor is not None else None
        es_mejor = self.mejor is None or t < self.mejor
        if es_mejor:
            self.mejor = t
            self.mejor_cum = cum
        if self.mejores_sec is None or len(self.mejores_sec) != len(secs):
            self.mejores_sec = secs[:]
        else:
            for i in range(len(secs)):
                if secs[i] < self.mejores_sec[i]:
                    self.mejores_sec[i] = secs[i]
        self.ultima = t
        self.ult_secs = secs
        self.t0 = ahora
        self.ult_cruce = ahora
        self.dist = 0.0
        self.sec_i = 0
        self.cum = []
        self.armado = False
        return [("vuelta", self.num, t, secs, es_mejor)]

    def actualizar(self, prev, cur, inc, ahora):
        ev = []
        if self.meta is None:
            return ev
        self.dist += inc
        t = time.ticks_diff(ahora, self.t0)
        lim = self.limites

        if lim and self.sec_i < len(lim) and self.dist >= lim[self.sec_i]:
            self.cum.append(t)
            self._delta(len(self.cum) - 1, t)
            self.sec_i += 1
            ev.append(("sector", self.sec_i, t))

        if lim is None and len(self.traza) < 600 and time.ticks_diff(ahora, self.t_traza) >= 1000:
            self.t_traza = ahora
            self.traza.append((int(self.dist), t))

        R = CFG["radio_meta_m"]
        d_act = dist_m(cur[0], cur[1], self.meta[0], self.meta[1])
        if self.armado and time.ticks_diff(ahora, self.ult_cruce) >= CFG["vuelta_min_s"] * 1000:
            cruza = d_act < R or (prev is not None and dist_seg(prev, cur, self.meta) < R)
            if cruza:
                ev.extend(self._cerrar(ahora))
                return ev
        if d_act > 2 * R:
            self.armado = True
        return ev

class Leds:
    COLORES = [(0, 40, 0), (0, 40, 0), (45, 35, 0), (45, 35, 0), (55, 0, 0)]

    def __init__(self):
        self.np = None
        self.ult = None
        if not CFG["leds"]:
            return
        try:
            from machine import Pin
            import neopixel
            self.np = neopixel.NeoPixel(Pin(CFG["led_pin"]), CFG["led_n"])
            self._pintar(0, False)
        except Exception as e:
            print("Sin LEDs:", e)
            self.np = None

    def _pintar(self, nivel, rojo_total):
        n = CFG["led_n"]
        mitad = n // 2
        for i in range(n):
            self.np[i] = (0, 0, 0)
        if rojo_total:
            for i in range(n):
                self.np[i] = (60, 0, 0)
        else:
            for i in range(min(nivel, mitad, len(self.COLORES))):
                self.np[i] = self.COLORES[i]
                self.np[n - 1 - i] = self.COLORES[i]
        self.np.write()

    def actualizar(self, rpm, ahora):
        if self.np is None:
            return
        try:
            lo = CFG["rpm_shift_min"]
            hi = CFG["rpm_max"]
            if rpm >= CFG["rpm_flash"]:
                estado = ("F", (ahora // 120) % 2)
            elif rpm <= lo:
                estado = ("N", 0)
            else:
                estado = ("N", min(5, 1 + int(5 * (rpm - lo) / max(hi - lo, 1))))
            if estado == self.ult:
                return
            self.ult = estado
            if estado[0] == "F":
                self._pintar(0, estado[1] == 1)
            else:
                self._pintar(estado[1], False)
        except Exception as e:
            print("Error LEDs:", e)

    def apagar(self):
        if self.np is not None:
            try:
                self._pintar(0, False)
            except Exception:
                pass

class Pantalla:
    FILAS = [("estado", 0xFFFFFF), ("vel", 0x00FF00), ("rpm", 0xFFFF00),
             ("gps", 0x00FFFF), ("vuelta", 0xFF9900), ("mejor", 0xFF9900),
             ("sect", 0xCCCCCC), ("sis", 0xFFFFFF), ("sd", 0x888888),
             ("msg", 0xFFFFFF)]

    def __init__(self):
        Widgets.fillScreen(0x111111)
        self.l = {}
        self.cache = {}
        self.msg_estilo = None
        self.bar_key = None
        y = 2
        for nombre, color in self.FILAS:
            self.l[nombre] = Widgets.Label("", 6, y, 1.0, color, 0x111111,
                                           Widgets.FONTS.DejaVu18)
            y += 22

    def restaurar(self):
        Widgets.fillScreen(0x111111)
        self.cache = {}
        self.bar_key = None

    def set(self, k, txt):
        txt = (txt + " " * 30)[:30]
        if self.cache.get(k) != txt:
            self.cache[k] = txt
            try:
                self.l[k].setText(txt)
            except Exception:
                pass

    def mensaje(self, texto, estilo):
        key = (texto, estilo)
        if self.cache.get("msg_key") == key:
            return
        self.cache["msg_key"] = key
        try:
            if estilo == 0:
                self.l["msg"].setColor(0x000000, 0xFFFF00)
            elif estilo == 1:
                self.l["msg"].setColor(0xFFFFFF, 0xCC0000)
            else:
                self.l["msg"].setColor(0xFFFFFF, 0x111111)
        except Exception:
            pass
        self.set("msg", texto)

    def barra(self, rpm):
        try:
            w = int(308 * min(1.0, max(0.0, rpm / float(CFG["rpm_max"]))))
            if rpm < CFG["rpm_shift_min"]:
                color = 0x00CC00
            elif rpm < CFG["rpm_flash"]:
                color = 0xFFCC00
            else:
                color = 0xFF0000
            key = (w // 3, color)
            if key == self.bar_key:
                return
            self.bar_key = key
            M5.Lcd.fillRect(6, 224, 308, 12, 0x222222)
            if w > 0:
                M5.Lcd.fillRect(6, 224, w, 12, color)
        except Exception:
            pass

class Teclado:
    # Páginas de caracteres dispuestas en exactamente DOS hileras de teclas grandes
    PAGINAS = [
        # Página 1: Minúsculas (A-N)
        [
            ["a", "b", "c", "d", "e", "f", "g"],
            ["h", "i", "j", "k", "l", "m", "n"]
        ],
        # Página 2: Minúsculas (O-Z, ñ, espacio)
        [
            ["o", "p", "q", "r", "s", "t", "u"],
            ["v", "w", "x", "y", "z", "ñ", " "]
        ],
        # Página 3: Mayúsculas (A-N)
        [
            ["A", "B", "C", "D", "E", "F", "G"],
            ["H", "I", "J", "K", "L", "M", "N"]
        ],
        # Página 4: Mayúsculas (O-Z, Ñ, espacio)
        [
            ["O", "P", "Q", "R", "S", "T", "U"],
            ["V", "W", "X", "Y", "Z", "Ñ", " "]
        ],
        # Página 5: Números y símbolos comunes
        [
            ["1", "2", "3", "4", "5", "6", "7"],
            ["8", "9", "0", "@", ".", "-", "_"]
        ],
        # Página 6: Símbolos especiales
        [
            ["!", "#", "$", "%", "&", "*", "?"],
            ["/", "=", "+", "(", ")", "[", "]"]
        ]
    ]

    def __init__(self, titulo, inicial="", clave=False, maximo=63):
        self.titulo = titulo
        self.texto = inicial[:maximo]
        self.clave = clave
        self.oculto = clave
        self.maximo = maximo
        self.pagina_actual = 0
        self.teclas = []

    def _construir(self):
        self.teclas = []
        filas = self.PAGINAS[self.pagina_actual]
        
        # Teclas grandes distribuidas en 2 hileras (7 columnas)
        cols = 7
        ancho_tecla = 42
        alto_tecla = 48
        espacio_x = 3
        espacio_y = 6
        x_inicio = 6
        y_inicio = 65

        for r in range(len(filas)):
            fila = filas[r]
            for c in range(len(fila)):
                char = fila[c]
                kx = x_inicio + c * (ancho_tecla + espacio_x)
                ky = y_inicio + r * (alto_tecla + espacio_y)
                self.teclas.append((kx, ky, ancho_tecla, alto_tecla, char))

        # Controles inferiores (Página Anterior, Página Siguiente, Borrar, OK, Cancelar, Ver)
        ctrl_y = 178
        ctrl_h = 46
        
        self.teclas.append((6, ctrl_y, 45, ctrl_h, "@prev"))
        self.teclas.append((55, ctrl_y, 45, ctrl_h, "@next"))
        self.teclas.append((104, ctrl_y, 50, ctrl_h, "@back"))
        
        if self.clave:
            self.teclas.append((158, ctrl_y, 50, ctrl_h, "@ver"))
            ok_x = 212
            ok_w = 60
            x_cancel = 276
            w_cancel = 38
        else:
            ok_x = 158
            ok_w = 95
            x_cancel = 257
            w_cancel = 57

        self.teclas.append((ok_x, ctrl_y, ok_w, ctrl_h, "@ok"))
        self.teclas.append((x_cancel, ctrl_y, w_cancel, ctrl_h, "@x"))

    def _etiqueta(self, v):
        if v == "@prev":
            return "< P"
        if v == "@next":
            return "P >"
        if v == "@back":
            return "Borrar"
        if v == "@ver":
            return "Ver" if self.oculto else "Ocu"
        if v == "@ok":
            return "OK"
        if v == "@x":
            return "X"
        if v == " ":
            return "esp"
        return v

    def _color(self, v):
        if v == "@ok":
            return 0x008800
        if v == "@x":
            return 0xAA0000
        if v in ("@prev", "@next"):
            return 0x0055AA
        if v.startswith("@") or v == " ":
            return 0x444444
        return 0x2C3E50

    def _dibujar_teclas(self):
        self._construir()
        lcd_rect(0, 56, 320, 184, 0x111111)
        
        # Indicador visual de página actual
        pag_txt = f"Pag {self.pagina_actual + 1}/{len(self.PAGINAS)}"
        lcd_texto(pag_txt, 10, 56, 0x00FFFF, 0x111111)

        for x, y, w, h, v in self.teclas:
            c = self._color(v)
            lcd_rect(x + 1, y + 1, w - 2, h - 2, c)
            lcd_borde(x, y, w, h, 0x666666)
            lcd_centro(self._etiqueta(v), x, y, w, h, 0xFFFFFF, c)

    def _dibujar_campo(self):
        lcd_rect(0, 2, 320, 52, 0x111111)
        lcd_texto(self.titulo[:32], 6, 2, 0x00FFFF)
        lcd_borde(4, 24, 312, 28, 0x888888)
        t = ("*" * len(self.texto)) if self.oculto else self.texto
        t += "_"
        while lcd_ancho(t) > 298 and len(t) > 1:
            t = t[1:]
        lcd_texto(t, 8, 29, 0xFFFFFF)

    def _buscar(self, x, y):
        for kx, ky, kw, kh, v in self.teclas:
            if kx <= x < kx + kw and ky <= y < ky + kh:
                return v
        return None

    def correr(self):
        lcd_limpiar()
        self._dibujar_campo()
        self._dibujar_teclas()
        while True:
            M5.update()
            t = detectar_toque()
            if t is not None:
                v = self._buscar(t[0], t[1])
                if v == "@ok":
                    return self.texto
                elif v == "@x":
                    return None
                elif v == "@back":
                    self.texto = self.texto[:-1]
                    self._dibujar_campo()
                elif v == "@prev":
                    self.pagina_actual = (self.pagina_actual - 1) % len(self.PAGINAS)
                    self._dibujar_teclas()
                elif v == "@next":
                    self.pagina_actual = (self.pagina_actual + 1) % len(self.PAGINAS)
                    self._dibujar_teclas()
                elif v == "@ver":
                    self.oculto = not self.oculto
                    self._dibujar_campo()
                    self._dibujar_teclas()
                elif v is not None:
                    if len(self.texto) < self.maximo:
                        self.texto += v
                        self._dibujar_campo()
            time.sleep_ms(15)

    c_prev = on if pagina > 0 else off

    for x, w, c, txt in ((2, 60, c_prev, "<"), (66, 60, c_next, ">"),

def elegir_red(redes, puede_salir):
    por = 5
    pagina = 0
    paginas = max(1, (len(redes) + por - 1) // por)
    dibujar = True
    while True:
        if dibujar:
            _dibujar_lista(redes, pagina, paginas, puede_salir)
            dibujar = False
        M5.update()
        t = detectar_toque()
        if t is not None:
            x, y = t
            if y < 26:
                if puede_salir and x >= 280:
                    return ("salir", None)
            elif y < 200:
                i = (y - 28) // 34
                idx = pagina * por + i
                if 0 <= i < por and idx < len(redes):
                    return ("red", redes[idx])
            else:
                if x < 64:
                    if pagina > 0:
                        pagina -= 1
                        dibujar = True
                elif x < 128:
                    if pagina < paginas - 1:
                        pagina += 1
                        dibujar = True
                elif x < 224:
                    return ("scan", None)
                else:
                    return ("manual", None)
        time.sleep_ms(15)


def conectar_wifi(ssid, pw, espera_s=15):
    w = E.wlan
    lcd_limpiar()
    lcd_centro("Conectando a", 0, 70, 320, 24)
    lcd_centro(ssid[:26], 0, 98, 320, 24, 0x00FFFF)
    try:
        w.disconnect()
    except Exception:
        pass
    time.sleep_ms(300)
    try:
        if pw:
            w.connect(ssid, pw)
        else:
            w.connect(ssid)
    except Exception as e:
        return False, "Error: %s" % e
    mala = getattr(network, "STAT_WRONG_PASSWORD", 202)
    sin_ap = getattr(network, "STAT_NO_AP_FOUND", 201)
    for i in range(int(espera_s * 5)):
        if w.isconnected():
            return True, ""
        try:
            st = w.status()
        except Exception:
            st = None
        if i > 10:
            if st == mala:
                return False, "Clave incorrecta"
            if st == sin_ap:
                return False, "Red no encontrada"
        lcd_centro(("." * (1 + i % 4)).ljust(4), 0, 140, 320, 24, 0xFFFF00)
        time.sleep_ms(200)
    return w.isconnected(), "Tiempo agotado"


def asistente_wifi(ui, puede_salir=True):
    E.pausa_enlace = True
    conectado = False
    try:
        time.sleep_ms(400)
        redes = None
        while True:
            if redes is None:
                redes = escanear_redes()
            accion, dato = elegir_red(redes, puede_salir)
            if accion == "salir":
                break
            if accion == "scan":
                redes = None
                continue
            if accion == "manual":
                ssid = Teclado("Nombre de la red (SSID)", "", False, 32).correr()
                if not ssid:
                    continue
                auth = 1
            else:
                ssid, _rssi, auth = dato
            pw = ""
            if auth != 0:
                ini = CFG["wifi_password"] if ssid == CFG["wifi_ssid"] and wifi_configurada() else ""
                pw = Teclado("Clave de " + ssid[:16], ini, True, 63).correr()
                if pw is None:
                    continue
            ok, motivo = conectar_wifi(ssid, pw)
            if ok:
                CFG["wifi_ssid"] = ssid
                CFG["wifi_password"] = pw
                guardado = guardar_wifi(ssid, pw)
                E.wifi_ok = True
                E.t_wifi = 0
                E.t_fallo = -100000
                if not E.ntp_hecho:
                    sincronizar_ntp()
                ip = ""
                try:
                    ip = E.wlan.ifconfig()[0]
                except Exception:
                    pass
                aviso("Conectado", ip + ("" if guardado else " (sin guardar)"),
                      0x00FF66, esperar=False, seg=1.5)
                conectado = True
                break
            aviso("No se pudo conectar", motivo, 0xFF5555)
    except Exception as e:
        print("Error en el asistente Wi-Fi:", e)
    finally:
        E.pausa_enlace = False
        ui.restaurar()
    return conectado

class Modem4G:
    def __init__(self):
        self.uart = None
        self.ppp = None
        self.t_intento = 0

    def _at(self, cmd, espera=b"OK", ms=3000):
        self.uart.write(cmd + "\r\n")
        t0 = time.ticks_ms()
        buf = b""
        while time.ticks_diff(time.ticks_ms(), t0) < ms:
            if self.uart.any():
                d = self.uart.read()
                if d:
                    buf += d
            if espera in buf:
                return True
            if b"ERROR" in buf:
                return False
            time.sleep_ms(20)
        return False

    def iniciar(self):
        self.t_intento = time.ticks_ms()
        try:
            from machine import UART
            self.uart = UART(CFG["modem_uart"], baudrate=CFG["modem_baud"],
                             tx=CFG["modem_tx"], rx=CFG["modem_rx"])
            if not (self._at("AT") and self._at("ATE0")):
                raise Exception("el modem no responde")
            self._at('AT+CGDCONT=1,"IP","%s"' % CFG["apn"])
            if not self._at("ATD*99#", b"CONNECT", 20000):
                raise Exception("no hay datos móviles")
            self.ppp = network.PPP(self.uart)
            self.ppp.active(True)
            self.ppp.connect()
            for _ in range(30):
                if self.ppp.isconnected():
                    print("4G conectado")
                    return True
                time.sleep(1)
        except Exception as e:
            print("4G fallo:", e)
        return False

    def conectado(self):
        try:
            return self.ppp is not None and self.ppp.isconnected()
        except Exception:
            return False

def estado_vivo(final=False):
    if final:
        return {"sesion_actual/activa": False, "sesion_actual/id": "",
                "sesion_actual/inicio": 0}
    v = {
        "sesion_actual/activa": True,
        "sesion_actual/id": E.id,
        "sesion_actual/inicio": E.inicio,
        "sesion_actual/ultimo_latido": timestamp_ms(),
        "sesion_actual/pendientes": E.cola.total(),
    }
    if E.ultimo_dato:
        v["sesion_actual/ultimo_dato"] = E.ultimo_dato
    if E.carrera:
        v["sesion_actual/carrera"] = E.carrera
    return v


def flush(final=False):
    origen, lote, info = E.cola.tomar(CFG["lote_max"])
    upd = {}
    for r in lote:
        upd[r["p"]] = r["d"]
    upd.update(estado_vivo(final))
    ack = E.ack
    if ack:
        upd["control/ack"] = ack
    ok, _ = firebase_request("PATCH", "", upd)
    if ok:
        E.cola.confirmar(origen, info)
        if ack is not None and E.ack is ack:
            E.ack = None
    else:
        E.cola.liberar()
    return ok, len(lote)


def finalizar_sesion(estado="finalizada"):
    if E.finalizada:
        return
    E.finalizada = True
    E.red_on = False
    t0 = time.ticks_ms()
    while E.red_viva and time.ticks_diff(time.ticks_ms(), t0) < 8000:
        time.sleep_ms(50)
    try:
        cerrar_sesion_en_cola(estado)
    except Exception:
        pass
    enviado = False
    try:
        if enlace_ok():
            for _ in range(15):
                ok, _n = flush(final=True)
                if not ok:
                    break
                if E.cola.total() == 0:
                    enviado = True
                    break
    except Exception as e:
        print("Error en cierre:", e)
    if not enviado:
        E.cola.volcar()
    print("Sesión finalizada" if enviado else "Sesión cerrada con pendientes en SD")

def gestionar_enlace():
    if E.pausa_enlace:
        return
    now = time.ticks_ms()
    if time.ticks_diff(now, E.t_wifi) < CFG["wifi_check_ms"]:
        return
    E.t_wifi = now
    try:
        E.wifi_ok = E.wlan.isconnected()
    except Exception:
        E.wifi_ok = False

    if E.wifi_ok:
        E.t_sin_wifi = None
    else:
        if E.t_sin_wifi is None:
            E.t_sin_wifi = now
        if time.ticks_diff(now, E.t_reintento) >= CFG["wifi_reintento_ms"]:
            E.t_reintento = now
            try:
                E.wlan.disconnect()
            except Exception:
                pass
            try:
                E.wlan.connect(CFG["wifi_ssid"], CFG["wifi_password"])
                print("Reintentando Wi-Fi...")
            except Exception as e:
                print("Error reconectando:", e)
        m = E.modem
        if (m is not None and not m.conectado()
                and time.ticks_diff(now, E.t_sin_wifi) > CFG["modem_espera_s"] * 1000
                and time.ticks_diff(now, m.t_intento) > 60000):
            m.iniciar()
    if E.modem is not None:
        E.modem_ok = E.modem.conectado()


def recuperar_sesion_anterior():
    ok, d = firebase_request("GET", "sesion_actual")
    if not ok:
        return False
    if isinstance(d, dict) and d.get("activa") is True:
        viejo = d.get("id")
        if viejo and viejo != E.id:
            ud = d.get("ultimo_dato")
            fin = d.get("ultimo_latido") or (ud.get("timestamp") if isinstance(ud, dict) else None) \
                or timestamp_ms()
            firebase_request("PATCH", "sesiones/" + viejo,
                             {"fin": fin, "estado": "interrumpida"})
            print("Sesión anterior marcada como interrumpida:", viejo)
    return True


def leer_control():
    ok, d = firebase_request("GET", "control")
    if not ok:
        E.t_fallo = time.ticks_ms()
        return
    if not isinstance(d, dict):
        d = {}
    cmd = d.get("comando")
    msg = d.get("mensaje")
    cid = cmd.get("id") if isinstance(cmd, dict) else None
    mid = msg.get("id") if isinstance(msg, dict) else None
    if not E.control_ok:
        E.ult_cmd = cid
        E.ult_msg = mid
        E.control_ok = True
        return
    if cid is not None and cid != E.ult_cmd:
        E.ult_cmd = cid
        with E.lock_in:
            E.entrada.append(("cmd", cid, cmd.get("accion", ""), cmd.get("valor")))
    if mid is not None and mid != E.ult_msg:
        E.ult_msg = mid
        with E.lock_in:
            E.entrada.append(("msg", mid, str(msg.get("texto", "")), None))


def ciclo_red():
    if E.pausa_enlace:
        return
    gestionar_enlace()
    if not enlace_ok():
        return
    now = time.ticks_ms()
    if time.ticks_diff(now, E.t_fallo) < 2000:
        return
    if not E.reloj_ok and time.ticks_diff(now, E.t_ntp) >= 30000:
        sincronizar_ntp()
    if not E.recuperada:
        if recuperar_sesion_anterior():
            E.recuperada = True
        else:
            E.intentos_rec += 1
            E.t_fallo = now
            if E.intentos_rec >= 3:
                E.recuperada = True
        return

    hay = E.cola.total() > 0
    backlog = E.cola.n_sd > 0
    if ((hay and (backlog or time.ticks_diff(now, E.t_subida) >= CFG["subida_ms"]))
            or time.ticks_diff(now, E.t_flush) >= CFG["latido_ms"]):
        E.t_subida = now
        ok, _n = flush()
        if ok:
            E.t_flush = now
        else:
            E.t_fallo = time.ticks_ms()
            return

    if time.ticks_diff(now, E.t_control) >= CFG["control_ms"]:
        E.t_control = now
        leer_control()


def tarea_red():
    E.red_viva = True
    try:
        while E.red_on:
            try:
                ciclo_red()
            except Exception as e:
                print("Error en hilo de red:", e)
                E.t_fallo = time.ticks_ms()
            time.sleep_ms(100)
    finally:
        E.red_viva = False

def marca_manual(origen):
    hora = hora_texto()
    ts = timestamp_ms()
    E.seq += 1
    E.cola.add("sesiones/%s/marcas/%013d_%05d" % (E.id, ts, E.seq),
               {"hora_marca": hora, "timestamp": ts, "origen": origen})
    mensaje("Marca " + hora, 2)


def fijar_meta_aqui():
    lat, lng, fuente = E.pos
    if fuente == "nofix":
        mensaje("Sin fix GPS", 3, 1)
        return
    E.vue.fijar_meta(lat, lng, time.ticks_ms())
    E.cola.add("sesiones/%s/meta" % E.id, {"lat": lat, "lng": lng})
    mensaje("Meta fijada", 3)


def ejecutar_comando(cid, accion, valor):
    ok = True
    try:
        if accion == "marca":
            marca_manual("box")
        elif accion == "frecuencia":
            E.muestreo_ms = max(200, min(10000, int(valor)))
        elif accion == "pit":
            E.pit = valor in (True, 1, "1", "on", "true", "si")
        elif accion == "limite_pit":
            CFG["pit_limite_kmh"] = int(valor)
        elif accion == "brillo":
            poner_brillo(valor)
        elif accion == "meta":
            fijar_meta_aqui()
        elif accion == "nueva_sesion":
            reiniciar_sesion()
        elif accion == "cerrar":
            E.activo = False
        elif accion == "apagar":
            E.apagar = True
            E.activo = False
        else:
            ok = False
    except Exception as e:
        print("Error en comando:", e)
        ok = False
    E.ack = {"id": cid, "accion": accion, "ok": ok, "timestamp": timestamp_ms()}


def procesar_entrada():
    while True:
        with E.lock_in:
            item = E.entrada.pop(0) if E.entrada else None
        if item is None:
            return
        if item[0] == "cmd":
            ejecutar_comando(item[1], item[2], item[3])
        else:
            mensaje(item[2], 12, 0)
            E.ack = {"id": item[1], "accion": "mensaje", "ok": True,
                     "timestamp": timestamp_ms()}


def procesar_eventos(eventos):
    vue = E.vue
    for ev in eventos:
        if ev[0] == "vuelta":
            _, num, t, secs, es_mejor = ev
            E.cola.add("sesiones/%s/vueltas/%03d" % (E.id, num), {
                "num": num, "tiempo_ms": t, "sectores": secs, "mejor": es_mejor,
                "timestamp": timestamp_ms()})
            E.cola.add("sesiones/%s/resumen" % E.id, {
                "vueltas": num, "mejor_ms": vue.mejor or 0,
                "teorica_ms": vue.teorica() or 0, "ultima_ms": t})
    if eventos:
        E.carrera = {
            "vuelta_actual": vue.num + 1,
            "completadas": vue.num,
            "ultima_ms": vue.ultima or 0,
            "mejor_ms": vue.mejor or 0,
            "teorica_ms": vue.teorica() or 0,
            "delta_ms": vue.delta if vue.delta is not None else 0,
        }

def pintar(ui, ahora, vel_kmh, rpm, bateria):
    vue = E.vue
    lat, lng, fuente = E.pos
    red = "WiFi" if E.wifi_ok else ("4G" if E.modem_ok else "SIN RED-toca")
    ui.set("estado", "En pista | " + red + (" | PIT" if E.pit else ""))
    ui.set("vel", "Velocidad: %.1f km/h" % vel_kmh)
    ui.set("rpm", "RPM: %d" % int(rpm))
    letra = {"real": "R", "sim": "S"}.get(fuente, "?")
    ui.set("gps", "GPS[%s] %.4f, %.4f" % (letra, lat, lng))
    ui.set("vuelta", "V%d  Ult %s" % (vue.num + 1, fmt_t(vue.ultima)))
    delta = ""
    if vue.delta is not None:
        delta = " D%+.2f" % (vue.delta / 1000.0)
    ui.set("mejor", "Mej %s%s" % (fmt_t(vue.mejor), delta))
    if vue.ult_secs:
        ui.set("sect", "S " + "/".join("%.1f" % (s / 1000.0) for s in vue.ult_secs))
    else:
        ui.set("sect", "S --")
    ui.set("sis", "Bat %d%%  Mem %dK  Pend %d" % (bateria, gc.mem_free() // 1024,
                                                   E.cola.total()))
    ui.set("sd", "SD %s  %s" % ("OK" if E.cola.sd else "NO", E.id[-12:]))

    if E.pit and vel_kmh > CFG["pit_limite_kmh"]:
        ui.mensaje("PIT! limite %d km/h" % CFG["pit_limite_kmh"], 1)
    elif E.msg_texto and time.ticks_diff(E.msg_hasta, ahora) > 0:
        ui.mensaje(E.msg_texto, E.msg_estilo)
    else:
        ui.mensaje("", -1)
    ui.barra(rpm)


def leer_bateria():
    try:
        return M5.Power.getBatteryLevel()
    except Exception:
        return 0


def bucle_principal(ui, leds, din):
    t_ult = time.ticks_ms()
    t_muestra = t_ult
    t_ui = 0
    t_gc = t_ult
    t_bat = 0
    bateria = leer_bateria()
    pos_prev = None
    vel_kmh = 0.0

    while E.activo:
        M5.update()
        ahora = time.ticks_ms()
        dt = time.ticks_diff(ahora, t_ult) / 1000.0
        t_ult = ahora
        if dt <= 0:
            dt = 0.001

        if M5.BtnA.wasPressed():
            marca_manual("boton")
        if M5.BtnB.wasPressed():
            E.apagar = True
            E.activo = False
            break
        if M5.BtnC.wasPressed():
            fijar_meta_aqui()
        tq = detectar_toque()
        if tq is not None and tq[1] < 24:
            if vel_kmh > 3:
                mensaje("WiFi: solo con el coche parado", 3, 1)
            else:
                asistente_wifi(ui, True)
                t_ult = time.ticks_ms()
                continue
        if E.entrada:
            procesar_entrada()

        gps_ms = None
        if E.gps is not None:
            E.gps.leer()
            if E.gps.valido():
                lat, lng, fuente = E.gps.lat, E.gps.lng, "real"
                gps_ms = E.gps.vel / 3.6
            else:
                lat, lng, fuente = E.pos[0], E.pos[1], "nofix"
        vel_ms = din.actualizar(dt, gps_ms)
        if E.gps is None:
            if CFG["sim_vel_kmh"] > 0:
                vel_ms = CFG["sim_vel_kmh"] / 3.6
            E.circ.avanzar(vel_ms * dt * CFG["sim_factor"])
            lat, lng = E.circ.latlng()
            fuente = "sim"
        vel_kmh = vel_ms * 3.6
        E.pos = (lat, lng, fuente)

        if fuente != "nofix":
            if pos_prev is None:
                pos_prev = (lat, lng)
            if E.vue.meta is None:
                E.vue.fijar_meta(lat, lng, ahora)
                E.cola.add("sesiones/%s/meta" % E.id, {"lat": lat, "lng": lng})
            inc = dist_m(pos_prev[0], pos_prev[1], lat, lng) if vel_kmh > 1.0 else 0.0
            eventos = E.vue.actualizar(pos_prev, (lat, lng), inc, ahora)
            pos_prev = (lat, lng)
            if eventos:
                procesar_eventos(eventos)

        leds.actualizar(din.rpm, ahora)

        if time.ticks_diff(ahora, t_muestra) >= E.muestreo_ms:
            t_muestra = ahora
            if time.ticks_diff(ahora, t_bat) >= 5000:
                t_bat = ahora
                bateria = leer_bateria()
            ts = timestamp_ms()
            E.seq += 1
            paquete = {
                "timestamp": ts,
                "aceleracion_x": round(din.a[0], 3),
                "aceleracion_y": round(din.a[1], 3),
                "aceleracion_z": round(din.a[2], 3),
                "latitud": round(lat, 6),
                "longitud": round(lng, 6),
                "velocidad": round(vel_kmh, 1),
                "rpm": int(din.rpm),
                "bateria_m5": bateria,
                "vuelta": E.vue.num + 1,
                "gps": fuente,
            }
            E.ultimo_dato = paquete
            E.cola.add("sesiones/%s/datos/%013d_%05d" % (E.id, ts, E.seq), paquete)

        if time.ticks_diff(ahora, t_ui) >= 200:
            t_ui = ahora
            pintar(ui, ahora, vel_kmh, din.rpm, bateria)

        if time.ticks_diff(ahora, t_gc) >= 10000:
            t_gc = ahora
            gc.collect()

        time.sleep_ms(10)


def main():
    global E
    M5.begin()
    ui = Pantalla()
    ui.set("estado", "Iniciando...")

    base = montar_sd()
    cargar_config()
    E = Estado()
    E.cola = Cola(base, base is not None)
    if CFG["modem_4g"]:
        E.modem = Modem4G()
    if CFG["gps_real"]:
        try:
            E.gps = GpsReal()
        except Exception as e:
            print("GPS no disponible:", e)
            E.gps = None
    leds = Leds()
    din = Dinamica()

    E.wlan = network.WLAN(network.STA_IF)
    E.wlan.active(True)
    if wifi_configurada():
        ui.set("estado", "Conectando WiFi...")
        try:
            E.wlan.connect(CFG["wifi_ssid"], CFG["wifi_password"])
        except Exception as e:
            print("Error Wi-Fi:", e)
        for _ in range(int(CFG["wifi_espera_s"])):
            if E.wlan.isconnected():
                break
            time.sleep(1)
    E.wifi_ok = E.wlan.isconnected()
    if not E.wifi_ok:
        asistente_wifi(ui, True)
        E.wifi_ok = E.wlan.isconnected()
    print("WiFi conectado:", E.wifi_ok)
    if E.wifi_ok and not E.ntp_hecho:
        sincronizar_ntp()
    ui.restaurar()

    ui.set("estado", "Calibrando IMU...")
    ui.set("vel", "No muevas el equipo")
    din.calibrar()

    E.circ = Circuito()
    nueva_sesion()

    try:
        _thread.stack_size(16 * 1024)
    except Exception:
        pass
    E.red_on = True
    _thread.start_new_thread(tarea_red, ())

    print("Iniciando sistema de telemetría...")
    estado_final = "finalizada"
    try:
        bucle_principal(ui, leds, din)
    except BaseException as e:
        try:
            sys.print_exception(e)
        except Exception:
            print("Error:", e)
    leds.apagar()
    finalizar_sesion(estado_final)
    if E.apagar:
        try:
            M5.Power.powerOff()
        except Exception as e:
            print("No se pudo apagar:", e)


if __name__ == "__main__":
    main()
